-- ======================================================
-- PRODUCTS TABLE
-- ======================================================
CREATE TABLE products (
    id TEXT PRIMARY KEY,  -- Shopify GID: gid://shopify/Product/123
    title TEXT NOT NULL,  -- Product title
    handle TEXT,          -- URL slug
    description TEXT,     -- Plain text description
    description_html TEXT,-- HTML description
    status TEXT,          -- ACTIVE, DRAFT, ARCHIVED
    vendor TEXT,          -- Vendor name
    product_type TEXT,    -- Product type/category
    tags TEXT[],          -- Shopify tags as array
    published_at TIMESTAMP WITH TIME ZONE,
    created_at TIMESTAMP WITH TIME ZONE,
    updated_at TIMESTAMP WITH TIME ZONE,
    total_inventory INTEGER,
    online_store_url TEXT,
    published_channels TEXT[]  -- e.g. ['online_store', 'pos']
);

-- ======================================================
-- PRODUCT VARIANTS TABLE
-- ======================================================
CREATE TABLE product_variants (
    id TEXT PRIMARY KEY,  -- Shopify GID: gid://shopify/ProductVariant/456
    product_id TEXT NOT NULL REFERENCES products(id) ON DELETE CASCADE,
    title TEXT,
    sku TEXT,
    price NUMERIC(10,2),
    compare_at_price NUMERIC(10,2),
    inventory_quantity INTEGER,
    barcode TEXT,
    weight NUMERIC(10,2),
    weight_unit TEXT,  -- GRAMS, KILOGRAMS, OUNCES, POUNDS
    requires_shipping BOOLEAN,
    taxable BOOLEAN,
    available_for_sale BOOLEAN,
    created_at TIMESTAMP WITH TIME ZONE,
    updated_at TIMESTAMP WITH TIME ZONE,
    inventory_item_id TEXT  -- Shopify internal link
);

-- ======================================================
-- PRODUCT IMAGES TABLE
-- ======================================================
CREATE TABLE product_images (
    id TEXT PRIMARY KEY,  -- Shopify GID: gid://shopify/MediaImage/321
    product_id TEXT NOT NULL REFERENCES products(id) ON DELETE CASCADE,
    url TEXT NOT NULL,
    alt_text TEXT,
    width INTEGER,
    height INTEGER,
    position INTEGER,
    created_at TIMESTAMP WITH TIME ZONE
);

-- ======================================================
-- VARIANT IMAGES TABLE
-- ======================================================
CREATE TABLE variant_images (
    variant_id TEXT NOT NULL REFERENCES product_variants(id) ON DELETE CASCADE,
    image_id TEXT NOT NULL REFERENCES product_images(id) ON DELETE CASCADE,
    PRIMARY KEY (variant_id, image_id)
);

-- ======================================================
-- VARIANT SELECTED OPTIONS
-- ======================================================
CREATE TABLE variant_selected_options (
    id SERIAL PRIMARY KEY,
    variant_id TEXT NOT NULL REFERENCES product_variants(id) ON DELETE CASCADE,
    option_name TEXT NOT NULL,
    option_value TEXT NOT NULL
);

-- ======================================================
-- PRODUCT OPTIONS
-- ======================================================
CREATE TABLE product_options (
    id SERIAL PRIMARY KEY,
    product_id TEXT NOT NULL REFERENCES products(id) ON DELETE CASCADE,
    name TEXT NOT NULL,
    position INTEGER,
    values TEXT[]
);

-- ======================================================
-- INVENTORY ITEMS
-- ======================================================
CREATE TABLE inventory_items (
    id TEXT PRIMARY KEY,  -- gid://shopify/InventoryItem/789
    sku TEXT,
    tracked BOOLEAN,
    requires_shipping BOOLEAN,
    created_at TIMESTAMP WITH TIME ZONE,
    updated_at TIMESTAMP WITH TIME ZONE
);

-- ======================================================
-- INVENTORY LEVELS
-- ======================================================
CREATE TABLE inventory_levels (
    id SERIAL PRIMARY KEY,
    inventory_item_id TEXT NOT NULL REFERENCES inventory_items(id) ON DELETE CASCADE,
    location_id TEXT NOT NULL,  -- gid://shopify/Location/101
    available_quantity INTEGER,
    updated_at TIMESTAMP WITH TIME ZONE,
    UNIQUE(inventory_item_id, location_id)
);

-- ======================================================
-- LOCATIONS TABLE
-- ======================================================
CREATE TABLE locations (
    id TEXT PRIMARY KEY,  -- gid://shopify/Location/101
    name TEXT NOT NULL,
    address1 TEXT,
    address2 TEXT,
    city TEXT,
    province TEXT,
    country TEXT,
    zip TEXT,
    active BOOLEAN
);

-- ======================================================
-- METAFIELDS TABLE
-- ======================================================
CREATE TABLE metafields (
    id TEXT PRIMARY KEY,  -- gid://shopify/Metafield/555
    owner_id TEXT NOT NULL,
    owner_type TEXT NOT NULL,  -- PRODUCT, VARIANT, etc.
    namespace TEXT NOT NULL,
    key TEXT NOT NULL,
    value TEXT,
    type TEXT,  -- single_line_text_field, number_integer, etc.
    created_at TIMESTAMP WITH TIME ZONE,
    updated_at TIMESTAMP WITH TIME ZONE
);

-- ======================================================
-- INDEXES
-- ======================================================
CREATE INDEX idx_products_handle ON products(handle);
CREATE INDEX idx_variants_product_id ON product_variants(product_id);
CREATE INDEX idx_variants_sku ON product_variants(sku);
CREATE INDEX idx_images_product_id ON product_images(product_id);
CREATE INDEX idx_variant_images_variant_id ON variant_images(variant_id);
CREATE INDEX idx_variant_selected_options_variant_id ON variant_selected_options(variant_id);
CREATE INDEX idx_product_options_product_id ON product_options(product_id);
CREATE INDEX idx_inventory_levels_item_id ON inventory_levels(inventory_item_id);
CREATE INDEX idx_inventory_levels_location_id ON inventory_levels(location_id);
CREATE INDEX idx_metafields_owner ON metafields(owner_id, owner_type);
